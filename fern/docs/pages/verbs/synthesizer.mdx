---
title: Synthesizer
subtitle: A **property** that can be used in a `say` verb to override the application default TTS settings.
---

## Parameters

<ParamField path="vendor" type="string" required={true}>
  Name of the TTS vendor, or 'default' to use the application default.
</ParamField>

<ParamField path="voice" type="string" required={true}>
  Name of the voice to use for the TTS, or 'default' to use the application default.
</ParamField>

<ParamField path="engine" type="string" required={false}>
  (Google specific) may be `standard`, `neural`, `generative`, or `long-form`.
</ParamField>

<ParamField path="gender" type="string" required={false}>
  (Google specific) may be `MALE`, `FEMALE`, `NEUTRAL`.
</ParamField>

<ParamField path="label" type="string" required={false}>
  Label associated with the TTS vendor.
</ParamField>

<ParamField path="language" type="string" required={false}>
  Language code for the TTS, or 'default' to use the application default.
</ParamField>

<ParamField path="options" type="object" required={false}>
  Vendor-specific options for the TTS, see below for supported properties.
</ParamField>

### Vendor-specific options

<AccordionGroup>

  <Accordion title="google">
    <ParamField path="model" type="string" required={false}>
      The model to use for text-to-speech synthesis. When specified, this enables Gemini TTS. Example: `gemini-2.5-flash-preview-tts`.
    </ParamField>
    <ParamField path="apiMode" type="string" required={false}>
      Controls which Google TTS API mode to use. Possible values:
      - `tts`: Standard Google Cloud TTS voices (default).
      - `live`: HD voices using streaming mode for higher quality output.
      - `gemini`: Gemini TTS for AI-powered speech synthesis.

      The mode is automatically selected based on configuration: if `options.model` is specified or a `model_id` is configured in the speech credentials, Gemini TTS is used. If an HD voice is selected, `live` mode is used. Otherwise, standard `tts` mode is used.
    </ParamField>
    <ParamField path="prompt" type="string" required={false}>
      A prompt sent to the TTS model to guide how the audio should be generated. Use this to specify the desired tone, emotion, speaking style, or context for the synthesized speech. This parameter is only applicable when using Gemini TTS (`apiMode: "gemini"` or when `model` is specified).
    </ParamField>
  </Accordion>

  <Accordion title="cartesia">
    <ParamField path="voice_mode" type="string" required={false}>
      `embedding` or `id` (see [Cartesia docs](https://docs.cartesia.ai/api-reference/tts/bytes#request.body.voice))
    </ParamField>
    <ParamField path="embedding" type="string" required={false}>
      a voice embedding (see [Cartesia docs](https://docs.cartesia.ai/api-reference/tts/bytes#request.body.voice.Ttsrequest%20Embedding%20Specifier.embedding))
    </ParamField>
    <ParamField path="emotion" type="string" required={false}>
      specifies emotion (see [Cartesia docs](https://docs.cartesia.ai/api-reference/tts/bytes#request.body.voice.Ttsrequest%20Embedding%20Specifier.__experimental_controls.emotion))
    </ParamField>
    <ParamField path="speed" type="number" required={false}>
      A number or named specifier (e.g "slow") (see [Cartesia docs](https://docs.cartesia.ai/api-reference/tts/bytes#request.body.voice.Ttsrequest%20Embedding%20Specifier.__experimental_controls.speed))
    </ParamField>
  </Accordion>
  
  <Accordion title="elevenlabs">
    <ParamField path="stability" type="number" required={false}>
       Defines the stability for voice settings (see [Elevenlabs docs](https://elevenlabs.io/docs/api-reference/websocket#properties))  
    </ParamField>
    <ParamField path="similarity_boost" type="number" required={false}>
       Defines the similarity boost for voice settings. (see [Elevenlabs docs](https://elevenlabs.io/docs/api-reference/websocket#properties))  
    </ParamField>
    <ParamField path="use_speaker_boost" type="boolean" required={false}>
       Defines the use speaker boost for voice settings. This parameter is available on V2+ models (see [Elevenlabs docs](https://elevenlabs.io/docs/api-reference/websocket#properties))  
    </ParamField>
    <ParamField path="style" type="number" required={false}>
       Defines the style for voice settings. This parameter is available on V2+ models. (see [Elevenlabs docs](https://elevenlabs.io/docs/api-reference/websocket#properties))  
    </ParamField>
    <ParamField path="model_id" type="string" required={false}>
       Identifier of the model that will be used (see [Elevenlabs docs](https://elevenlabs.io/docs/api-reference/websocket#properties))  
    </ParamField>
  </Accordion>

  <Accordion title="rimelabs">
    <ParamField path="pauseBetweenBrackets" type="boolean" required={false}>
      When set to true, adds pauses between words enclosed in angle brackets. The number inside the brackets specifies the pause duration in milliseconds.
Example: "Hi. \<200> I'd love to have a conversation with you." adds a 200ms pause between the first and second sentences. see [Rimelabs docs](https://docs.rime.ai/api-reference/endpoint/streaming-pcm)
    </ParamField>
    <ParamField path="phonemizeBetweenBrackets" type="boolean" required={false}>
      When set to true, you can specify the phonemes for a word enclosed in curly brackets. see [Rimelabs docs](https://docs.rime.ai/api-reference/endpoint/streaming-pcm)
    </ParamField>
    <ParamField path="inlineSpeedAlpha" type="string" required={false}>
      Comma-separated list of speed values applied to words in square brackets. Values < 1.0 speed up speech, > 1.0 slow it down. Example: "This sentence is [really] [fast]" with inlineSpeedAlpha "0.5, 3" will make "really" slow and "fast" fast. see [Rimelabs docs](https://docs.rime.ai/api-reference/endpoint/streaming-pcm)
    </ParamField>
    <ParamField path="speedAlpha" type="number" required={false}>
      Adjusts the speed of speech. Lower than 1.0 is faster than default. Higher than 1.0 is slower than default. see [Rimelabs docs](https://docs.rime.ai/api-reference/endpoint/streaming-pcm)
    </ParamField>
    <ParamField path="reduceLatency" type="boolean" required={false}>
      Reduces the latency of response, at the cost of some possible mispronunciation of digits and abbreviations. see [Rimelabs docs](https://docs.rime.ai/api-reference/endpoint/streaming-pcm)
    </ParamField>
  </Accordion>

  <Accordion title="murf">
    <ParamField path="style" type="string" required={false}>
      The speaking style for the voice, e.g. `Conversational` (see [Murf docs](https://murf.ai/api/docs/text-to-speech/web-sockets))
    </ParamField>
    <ParamField path="rate" type="number" required={false}>
      Adjusts the speaking rate. (see [Murf docs](https://murf.ai/api/docs/text-to-speech/web-sockets))
    </ParamField>
    <ParamField path="pitch" type="number" required={false}>
      Adjusts the voice pitch. (see [Murf docs](https://murf.ai/api/docs/text-to-speech/web-sockets))
    </ParamField>
    <ParamField path="variation" type="number" required={false}>
      Controls the amount of variation in pace, pitch and emphasis. (see [Murf docs](https://murf.ai/api/docs/text-to-speech/web-sockets))
    </ParamField>
  </Accordion>

  <Accordion title="xai">
    Voices: `eve` (default), `ara`, `leo`, `rex`, `sal`.

    Supported languages: `auto`, `en`, `ar-EG`, `ar-SA`, `ar-AE`, `bn`, `zh`, `fr`, `de`, `hi`, `id`, `it`, `ja`, `ko`, `pt-BR`, `pt-PT`, `ru`, `es-MX`, `es-ES`, `tr`, `vi`.

    <ParamField path="speed" type="number" required={false}>
      Controls the speaking speed. A number between 0.7 and 1.5, inclusive. Defaults to 1.0.
    </ParamField>

    <ParamField path="optimize_streaming_latency" type="number" required={false}>
      Latency optimization level. 0 (default, no optimization), 1, or 2 (max latency reduction).
    </ParamField>

    <ParamField path="text_normalization" type="boolean" required={false}>
      Enable text normalization (expands numbers, dates, abbreviations for more natural speech). Defaults to false.
    </ParamField>

    <ParamField path="with_timestamps" type="boolean" required={false}>
      Return word-level timestamps alongside audio. Supported on the STREAMING TTS path only (ignored for one-shot `say` synthesis). Defaults to false.
    </ParamField>

    Note: xAI TTS supports both one-shot synthesis (via the `say` verb) and low-latency streaming TTS. It uses the same xAI API key credential that is used for xAI STT.
  </Accordion>

  <Accordion title="whisper">
    <ParamField path="model_id" type="string" required={false}>
      TTS model to use. (see [Whisper docs](https://platform.openai.com/docs/api-reference/audio/createSpeech))
    </ParamField>
  </Accordion>

  <Accordion title="kugelaudio">
    <ParamField path="speed" type="number" required={false}>
      Speaking rate. Defaults to `1.0`. (see [KugelAudio](/guides/features/tts-vendor-settings/kugel-audio))
    </ParamField>
    <ParamField path="cfg_scale" type="number" required={false}>
      Guidance scale. Defaults to `2.0`.
    </ParamField>
    <ParamField path="temperature" type="number" required={false}>
      Sampling temperature.
    </ParamField>
    <ParamField path="normalize" type="boolean" required={false}>
      Expand numbers, dates and abbreviations before synthesis. Defaults to `true`.
    </ParamField>
    <ParamField path="project_id" type="number" required={false}>
      KugelAudio project id.
    </ParamField>
    <ParamField path="dictionary_ids" type="array" required={false}>
      Pronunciation dictionary ids, as an array of numbers.
    </ParamField>
    <ParamField path="api_uri" type="string" required={false}>
      KugelAudio host, e.g. `api.eu.kugelaudio.com`. Defaults to `api.kugelaudio.com`.
    </ParamField>
  </Accordion>

  <Accordion title="speechify">
    <ParamField path="loudness_normalization" type="boolean" required={false}>
      Normalize output loudness. Defaults to `false`. (see [Speechify](/guides/features/tts-vendor-settings/speechify))
    </ParamField>
    <ParamField path="text_normalization" type="boolean" required={false}>
      Expand numbers, dates and abbreviations before synthesis. Defaults to `false`.
    </ParamField>
    <ParamField path="api_uri" type="string" required={false}>
      Speechify host. Defaults to `api.speechify.ai`.
    </ParamField>
  </Accordion>
